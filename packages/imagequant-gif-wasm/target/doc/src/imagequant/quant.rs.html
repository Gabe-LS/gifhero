31    pub(crate) fn new(attr: &Attributes, hist: HistogramInternal, freeze_result_colors: bool, gamma: f64) -> Result<Self, Error> {
32        if attr.progress(f32::from(attr.progress_stage1)) { return Err(Aborted); }
33        let (max_mse, target_mse, target_mse_is_zero) = attr.target_mse(hist.items.len());
34        let (mut palette, palette_error) = find_best_palette(attr, target_mse, target_mse_is_zero, max_mse, hist)?;
35        if freeze_result_colors {
36            palette.iter_mut().for_each(|(_, p)| *p = p.to_fixed());
37        }
38        if attr.progress(f32::from(attr.progress_stage3).mul_add(0.95, f32::from(attr.progress_stage1) + f32::from(attr.progress_stage2))) {
39            return Err(Aborted);
40        }
41        if let (Some(palette_error), Some(max_mse)) = (palette_error, max_mse) {
42            if palette_error > max_mse {
43                attr.verbose_print(format!(
44                    "  image degradation MSE={:0.3} (Q={}) exceeded limit of {:0.3} ({})",
45                    mse_to_standard_mse(palette_error),
46                    mse_to_quality(palette_error),
47                    mse_to_standard_mse(max_mse),
48                    mse_to_quality(max_mse)
49                ));
50                return Err(QualityTooLow);
51            }
52        }
53
54        sort_palette(attr, &mut palette);
55
56        Ok(Self {
57            palette,
58            gamma,
59            palette_error,
60            min_posterization_output: attr.min_posterization(),
61            use_dither_map: attr.use_dither_map,
62            remapped: None,
63            progress_callback: None,
64            int_palette: Palette {
65                count: 0,
66                entries: [RGBA::default(); MAX_COLORS],
67            },
68            dither_level: 1.,
69            single_threaded_dithering: attr.single_threaded_dithering,
70        })
71    }

82    #[inline(never)]
83    pub(crate) fn write_remapped_image_rows_internal(&mut self, image: &mut Image, mut output_pixels: RowBitmapMut<'_, MaybeUninit<PalIndexRemap>>) -> Result<(), Error> {
84        let progress_stage1 = if self.use_dither_map != DitherMapMode::None { 20 } else { 0 };
85        if self.remap_progress(progress_stage1 as f32 * 0.25) {
86            return Err(Error::Aborted);
87        }
88
89        image.free_histogram_inputs();
90
91        let mut palette = self.palette.clone();
92        let mut remapped = Box::new(Remapped {
93            int_palette: Palette { count: 0, entries: [RGBA::default(); MAX_COLORS] },
94            palette_error: None,
95        });
96        if self.dither_level == 0. {
97            palette.init_int_palette(&mut remapped.int_palette, self.gamma, self.min_posterization_output);
98            remapped.palette_error = Some(remap_to_palette(&mut image.px, image.background.as_deref_mut(), image.importance_map.as_deref(), &mut output_pixels, &mut palette)?.0);
99        } else {
100            let uses_background = image.background.is_some();
101            let dither_map_error = Self::optionally_generate_dither_map(self.use_dither_map, image, uses_background, &mut output_pixels, &mut palette)?;
102            if self.remap_progress(progress_stage1 as f32 * 0.5) {
103                return Err(Error::Aborted);
104            }
105
106            let output_image_is_remapped = dither_map_error.is_some();
107            let palette_error = dither_map_error.or(self.palette_error);
108
109            // remapping above was the last chance to do K-Means iteration, hence the final palette is set after remapping
110            palette.init_int_palette(&mut remapped.int_palette, self.gamma, self.min_posterization_output);
111            remapped.palette_error = palette_error;
112            let max_dither_error = (palette_error.unwrap_or(quality_to_mse(80)) * 2.4).max(quality_to_mse(35)) as f32;
113            remap_to_palette_floyd(image, output_pixels, &palette, self, max_dither_error, output_image_is_remapped)?;
114        }
115        self.remapped = Some(remapped);
116        Ok(())
117    }

119    fn optionally_generate_dither_map(use_dither_map: DitherMapMode, image: &mut Image<'_>, uses_background: bool, output_pixels: &mut RowBitmapMut<'_, MaybeUninit<PalIndexRemap>>, palette: &mut PalF) -> Result<Option<f64>, Error> {
120        let is_image_huge = (image.px.width * image.px.height) > 2000 * 2000;
121        let allow_dither_map = use_dither_map == DitherMapMode::Always || (!is_image_huge && use_dither_map != DitherMapMode::None);
122        let generate_dither_map = allow_dither_map && image.dither_map.is_none();
123        if !generate_dither_map {
124            return Ok(None);
125        }
126
127        // If dithering (with dither map) is required, this image is used to find areas that require dithering
128        let (palette_error, row_pointers_remapped) = remap_to_palette(&mut image.px, None, image.importance_map.as_deref(), output_pixels, palette)?;
129        image.update_dither_map(&row_pointers_remapped, &*palette, uses_background)?;
130        Ok(Some(palette_error))
131    }

237    /// Remap image into an existing buffer. Use [`remapped()`][Self::remapped] if you don't have a pre-allocated buffer to reuse.
238    ///
239    /// Writes 1-byte-per-pixel uncompressed bitmap into the `Vec`.
240    ///
241    /// Returns the palette.
242    #[inline]
243    pub fn remap_into_vec(&mut self, image: &mut Image<'_>, buf: &mut Vec<PalIndexRemap>) -> Result<Vec<RGBA>, Error> {
244        let len = image.width() * image.height();
245        // Capacity is essential here, as it creates uninitialized buffer
246        unsafe {
247            buf.clear();
248            buf.try_reserve_exact(len)?;
249            self.remap_into(image, &mut buf.spare_capacity_mut()[..len])?;
250            buf.set_len(len);
251        }
252        Ok(self.palette_vec())
253    }

255    /// Remap image into an existing buffer.
256    ///
257    /// This is a low-level call for use when existing memory has to be reused. Use [`remapped()`][Self::remapped] or [`remap_into_vec()`][Self::remap_into_vec] if possible.
258    ///
259    /// Writes 1-byte-per-pixel uncompressed bitmap into the pre-allocated buffer.
260    ///
261    /// You should call [`palette()`][Self::palette] _after_ this call, but not before it,
262    /// because remapping refines the palette.
263    #[inline]
264    pub fn remap_into(&mut self, image: &mut Image<'_>, output_buf: &mut [MaybeUninit<PalIndexRemap>]) -> Result<(), Error> {
265        let required_size = (image.width()) * (image.height());
266        let output_buf = output_buf.get_mut(0..required_size).ok_or(BufferTooSmall)?;
267
268        let rows = RowBitmapMut::new_contiguous(output_buf, image.width());
269        self.write_remapped_image_rows_internal(image, rows)
270    }

331fn sort_palette(attr: &Attributes, palette: &mut PalF) {
332    let last_index_transparent = attr.last_index_transparent;
333
334    let mut tmp: ArrayVec<_, { MAX_COLORS }> = palette.iter_mut().map(|(c, p)| (*c, *p)).collect();
335    tmp.sort_by_key(|(color, pop)| {
336        let is_transparent = color.a <= MAX_TRANSP_A;
337        (is_transparent == last_index_transparent, Reverse(OrdFloat::new(pop.popularity())))
338    });
339    palette.iter_mut().zip(tmp).for_each(|((dcol, dpop), (scol, spop))| {
340        *dcol = scol;
341        *dpop = spop;
342    });
343
344    if last_index_transparent {
345        let alpha_index = palette.as_slice().iter().enumerate()
346            .filter(|(_, c)| c.a <= MAX_TRANSP_A)
347            .min_by_key(|(_, c)| OrdFloat::new(c.a))
348            .map(|(i, _)| i);
349        if let Some(alpha_index) = alpha_index {
350            let last_index = palette.as_slice().len() - 1;
351            palette.swap(last_index, alpha_index);
352        }
353    } else {
354        let num_transparent = palette.as_slice().iter().enumerate()
355            .filter(|(_, c)| c.a <= MAX_TRANSP_A)
356            .map(|(i, _)| i + 1) // num entries, not index
357            .max();
358        if let Some(num_transparent) = num_transparent {
359            attr.verbose_print(format!("  eliminated opaque tRNS-chunk entries...{} entr{} transparent", num_transparent, if num_transparent == 1 { "y" } else { "ies" }));
360        }
361    }
362}

371/// Repeats mediancut with different histogram weights to find palette with minimum error.
372///
373///  `feedback_loop_trials` controls how long the search will take. < 0 skips the iteration.
374#[allow(clippy::or_fun_call)]
375pub(crate) fn find_best_palette(attr: &Attributes, target_mse: f64, target_mse_is_zero: bool, max_mse: Option<f64>, mut hist: HistogramInternal) -> Result<(PalF, Option<f64>), Error> {
376    // hist.items includes fixed colors already
377    let few_input_colors = hist.items.len() <= attr.max_colors as usize;
378    // actual target_mse passed to this method has extra diff from posterization
379    if few_input_colors && target_mse_is_zero {
380        return Ok(palette_from_histogram(&hist, attr.max_colors));
381    }
382
383    let mut max_colors = attr.max_colors;
384    let total_trials = attr.feedback_loop_trials(hist.items.len()) as i16;
385    let mut trials_left = total_trials;
386    let mut best_palette = None;
387    let mut target_mse_overshoot = if total_trials > 0 { 1.05 } else { 1. };
388    let mut fails_in_a_row = 0;
389    let mut palette_error = None;
390    let mut palette = loop {
391        let max_mse_per_color = target_mse.max(palette_error.unwrap_or(quality_to_mse(1))).max(quality_to_mse(51)) * 1.2;
392        let mut new_palette = mediancut(&mut hist, max_colors, target_mse * target_mse_overshoot, max_mse_per_color)?
393            .with_fixed_colors(attr.max_colors, &hist.fixed_colors);
394
395        let stage_done = (f32::from(trials_left.max(0)) / f32::from(total_trials + 1)).mul_add(-(f32::from(trials_left.max(0)) / f32::from(total_trials + 1)), 1.);
396        let overall_done = stage_done.mul_add(f32::from(attr.progress_stage2), f32::from(attr.progress_stage1));
397        attr.verbose_print(format!("  selecting colors...{}%", (100. * stage_done) as u8));
398
399        if trials_left <= 0 { break Some(new_palette); }
400
401        let first_run_of_target_mse = best_palette.is_none() && target_mse > 0.;
402        let total_error = Kmeans::iteration(&mut hist, &mut new_palette, !first_run_of_target_mse)?;
403        if best_palette.is_none() || total_error < palette_error.unwrap_or(f64::MAX) || (total_error <= target_mse && new_palette.len() < max_colors as usize) {
404            if total_error < target_mse && total_error > 0. {
405                target_mse_overshoot = if (target_mse_overshoot * 1.25) < (target_mse / total_error) {target_mse_overshoot * 1.25 } else {target_mse / total_error }; // if number of colors could be reduced, try to keep it that way
406            }
407            palette_error = Some(total_error);
408            max_colors = max_colors.min(new_palette.len() as PalLen + 1);
409            trials_left -= 1;
410            fails_in_a_row = 0;
411            best_palette = Some(new_palette);
412        } else {
413            fails_in_a_row += 1;
414            target_mse_overshoot = 1.;
415            trials_left -= 5 + fails_in_a_row;
416        }
417        if attr.progress(overall_done) || trials_left <= 0 {
418            break best_palette;
419        }
420    }.ok_or(ValueOutOfRange)?;
421
422    refine_palette(&mut palette, attr, &mut hist, max_mse, &mut palette_error)?;
423
424    Ok((palette, palette_error))
425}

428fn refine_palette(palette: &mut PalF, attr: &Attributes, hist: &mut HistogramInternal, max_mse: Option<f64>, palette_error: &mut Option<f64>) -> Result<(), Error> {
429    let (iterations, iteration_limit) = attr.kmeans_iterations(hist.items.len(), palette_error.is_some());
430    if iterations > 0 {
431        attr.verbose_print("  moving colormap towards local minimum");
432        let mut i = 0;
433        while i < iterations {
434            let stage_done = f32::from(i) / f32::from(iterations);
435            let overall_done = (stage_done * f32::from(attr.progress_stage3)).mul_add(0.89, f32::from(attr.progress_stage1) + f32::from(attr.progress_stage2));
436            if attr.progress(overall_done) {
437                break;
438            }
439
440            let pal_err = Kmeans::iteration(hist, palette, false)?;
441            debug_assert!(pal_err < 1e20);
442            let previous_palette_error = *palette_error;
443            *palette_error = Some(pal_err);
444
445            if let Some(previous_palette_error) = previous_palette_error {
446                if (previous_palette_error - pal_err).abs() < iteration_limit {
447                    break;
448                }
449            }
450            i += if pal_err > max_mse.unwrap_or(1e20) * 1.5 { 2 } else { 1 };
451        }
452    }
453    Ok(())
454}

457fn palette_from_histogram(hist: &HistogramInternal, max_colors: PalLen) -> (PalF, Option<f64>) {
458    let mut hist_pal = PalF::new();
459    for item in hist.items.iter() {
460        hist_pal.push(item.color, PalPop::new(item.perceptual_weight));
461    }
462
463    (hist_pal.with_fixed_colors(max_colors, &hist.fixed_colors), Some(0.))
464}

466pub(crate) fn quality_to_mse(quality: u8) -> f64 {
467    if quality == 0 {
468        return 1e20; // + epsilon for floating point errors
469    }
470    if quality >= 100 { return 0.; }
471    let extra_low_quality_fudge = (0.016 / (0.001 + f64::from(quality)) - 0.001).max(0.);
472    LIQ_WEIGHT_MSE * (extra_low_quality_fudge + 2.5 / (210. + f64::from(quality)).powf(1.2) * (100.1 - f64::from(quality)) / 100.)
473}
