10/// K-Means iteration: new palette color is computed from weighted average of colors that map best to that palette entry.
11// avoid false sharing
12pub(crate) struct Kmeans {
13    averages: Vec<ColorAvg>,
14    weighed_diff_sum: f64,
15}
16
17#[derive(Copy, Clone, Default)]
18struct ColorAvg {
19    pub sum: Argb<f64>,
20    pub total: f64,
21}
22
23impl Kmeans {
24    #[inline]
25    pub fn new(pal_len: usize) -> Result<Self, Error> {
26        let mut averages = Vec::new();
27        averages.try_reserve_exact(pal_len)?;
28        averages.resize(pal_len, ColorAvg::default());
29        Ok(Self {
30            averages,
31            weighed_diff_sum: 0.,
32        })
33    }
34
35    #[inline]
36    pub fn update_color(&mut self, px: f_pixel, value: f32, matched: PalIndex) {
37        let c = &mut self.averages[matched as usize];
38        c.sum += (px.0 * value).map(f64::from);
39        c.total += f64::from(value);
40    }
41
42    pub fn finalize(self, palette: &mut PalF) -> f64 {
43        for (avg, (color, pop)) in self.averages.iter().zip(palette.iter_mut()).filter(|(_, (_, pop))| !pop.is_fixed()) {
44            let total = avg.total;
45            *pop = PalPop::new(total as f32);
46            if total > 0. && color.a != 0. {
47                *color = avg.sum.map(move |c| (c / total) as f32).into();
48            }
49        }
50        self.weighed_diff_sum
51    }
52
53    #[inline(never)]
54    pub(crate) fn iteration(hist: &mut HistogramInternal, palette: &mut PalF, adjust_weight: bool) -> Result<f64, Error> {
55        if hist.items.is_empty() {
56            return Ok(0.);
57        }
58
59        let n = Nearest::new(palette)?;
60        let colors = palette.as_slice();
61        let len = colors.len();
62
63        let tls = ThreadLocal::new();
64        let total = hist.total_perceptual_weight;
65
66        // chunk size is a trade-off between parallelization and overhead
67        hist.items.par_chunks_mut(256).for_each({
68            let tls = &tls;
69            move |batch| {
70                let kmeans = tls.get_or(move || CacheLineAlign(RefCell::new(Self::new(len))));
71                if let Ok(ref mut kmeans) = *kmeans.0.borrow_mut() {
72                    kmeans.iterate_batch(batch, &n, colors, adjust_weight);
73                }
74            }
75        });
76
77        let diff = tls.into_iter()
78            .map(|c| c.0.into_inner())
79            .reduce(Self::try_merge)
80            .transpose()?
81            .map_or(0., |kmeans| kmeans.finalize(palette) / total);
82
83        replace_unused_colors(palette, hist)?;
84        Ok(diff)
85    }
86
87    fn iterate_batch(&mut self, batch: &mut [HistItem], n: &Nearest, colors: &[f_pixel], adjust_weight: bool) {
88        self.weighed_diff_sum += batch.iter_mut().map(|item| {
89            let px = item.color;
90            let (matched, mut diff) = n.search(&px, item.likely_palette_index());
91            item.tmp.likely_palette_index = matched;
92            if adjust_weight {
93                let remapped = colors[matched as usize];
94                let (_, new_diff) = n.search(&f_pixel(px.0 + px.0 - remapped.0), matched);
95                diff = new_diff;
96                item.adjusted_weight = 2.0f32.mul_add(item.adjusted_weight, item.perceptual_weight) * (0.5 + diff);
97            }
98            debug_assert!(f64::from(diff) < 1e20);
99            self.update_color(px, item.adjusted_weight, matched);
100            f64::from(diff * item.perceptual_weight)
101        }).sum::<f64>();
102    }
103
104    #[inline]
105    pub fn merge(mut self, new: Self) -> Self {
106        self.weighed_diff_sum += new.weighed_diff_sum;
107        self.averages.iter_mut().zip(new.averages).for_each(|(p, n)| {
108            p.sum += n.sum;
109            p.total += n.total;
110        });
111        self
112    }
113
114    #[inline]
115    pub fn try_merge<E>(old: Result<Self, E>, new: Result<Self, E>) -> Result<Self, E> {
116        match (old, new) {
117            (Ok(old), Ok(new)) => Ok(Self::merge(old, new)),
118            (Err(e), _) | (_, Err(e)) => Err(e),
119        }
120    }
121}

123/// kmeans may have merged or obsoleted some palette entries.
124/// This replaces these entries with histogram colors that are currently least-fitting the palette.
125fn replace_unused_colors(palette: &mut PalF, hist: &HistogramInternal) -> Result<(), Error> {
126    for pal_idx in 0..palette.len() {
127        let Some(pop) = palette.pop_as_slice().get(pal_idx) else { break };
128        if pop.popularity() == 0. && !pop.is_fixed() {
129            let n = Nearest::new(palette)?;
130            let mut worst = None;
131            let mut worst_diff = 0.;
132            let colors = palette.as_slice();
133            // the search is just for diff, ignoring adjusted_weight,
134            // because the palette already optimizes for the max weight, so it'd likely find another redundant entry.
135            for item in hist.items.iter() {
136                // the early reject avoids running full palette search for every entry
137                let may_be_worst = colors.get(item.likely_palette_index() as usize)
138                    .map_or(true, |pal| pal.diff(&item.color) > worst_diff);
139                if may_be_worst {
140                    let diff = n.search(&item.color, item.likely_palette_index()).1;
141                    if diff > worst_diff {
142                        worst_diff = diff;
143                        worst = Some(item);
144                    }
145                }
146            }
147            if let Some(worst) = worst {
148                palette.set(pal_idx, worst.color, PalPop::new(worst.adjusted_weight));
149            }
150        }
151    }
152    Ok(())
153}
